% Truncating before merging gives a wrong answer (deck, the correctness condition of README 2.2 and 2.4).
% The same Pauli string lives on two GPUs with the coefficients 0.7e-4 and 0.6e-4; the cutoff is 1e-4.
%   1  truncate on each GPU: both pieces are below the cutoff and are dropped, the term is lost
%   2  merge, then truncate: the pieces add to 1.3e-4, which is above the cutoff, the term is kept
% The two coefficients and the cutoff are an illustration (the cutoff is that of the measured setting).
% Bottom row, measured: kicked Ising with 22.5 million terms at cutoff 1e-4; the terms are divided over the
% GPUs and every GPU computes to the end without merging with the others: 0.1675 on 4 GPUs, 0.189 on 16 GPUs,
% against 0.1531 for the run that merges before it truncates (README 2.4 and the table of 4.9).

  \tikzset{rc/.style={draw=black!70, line width=0.35pt, anchor=north west, inner sep=0pt, minimum height=0.42cm,
                      text height=0.26cm, text depth=0.08cm, font=\rmfamily\footnotesize},
           tr/.style={rc, minimum height=0.48cm, text height=0.28cm, text depth=0.08cm},
           mc/.style={rc, minimum height=0.5cm, text height=0.28cm, text depth=0.09cm}}
  % ================= the same term on two GPUs =================
  \node[note] at (1.55,-1.2) {Cutoff = 1e-4};
  \foreach \g/\y/\c in {0/-1.575/0.7e-4, 1/-3.375/0.6e-4}
  {
    \node[gpu, anchor=north west, minimum width=3.1cm, minimum height=1.5cm] at (0,\y) {};
    \node[cttl, anchor=north] at (1.55,{\y-0.04}) {GPU \g};
    \node[hcell, minimum width=1.25cm] at (0.2,{\y-0.52}) {Coefficient};
    \node[hcell, minimum width=1.45cm] at (1.45,{\y-0.52}) {Pauli String};
    \node[rc, fill=white, minimum width=1.25cm] at (0.2,{\y-0.84}) {\c};
    \node[rc, fill=sa, minimum width=1.45cm] at (1.45,{\y-0.84}) {X Z Y I};
  }
  % both rows reach both paths
  \draw[line width=0.8pt] (3.5,-2.325) -- (3.5,-4.125);
  \draw[arr] (3.1,-2.325) -- (3.5,-2.325) -- (3.5,-1.7) -- (3.98,-1.7);
  \draw[arr] (3.1,-4.125) -- (3.5,-4.125) -- (3.5,-4.75) -- (3.98,-4.75);
  % ================= 1: truncate on each GPU =================
  \node[frame, anchor=north west, minimum width=9.9cm, minimum height=2.75cm] at (4.0,0) {};

  \node[lttl, anchor=west] at (4.6,-0.42) {Truncate on Each GPU};

    \node[tr, fill=cyanp, minimum width=1.1cm] at (4.35,\y) {GPU \g};
    \node[tr, fill=badc, minimum width=1.0cm] at (5.45,\y) {\c};
    \node[tr, fill=badc, minimum width=1.3cm] at (6.45,\y) {X Z Y I};
    \node[tr, fill=badc, minimum width=0.4cm, text=dred] at (7.75,\y) {\ding{55}};
  }
  \draw[arr] (8.35,-1.7) -- (10.92,-1.7);
  \node[note, anchor=south] at (9.63,-1.67) {Both Below the Cutoff};
  \node[draw=dred, line width=0.9pt, fill=badc, anchor=north west, minimum width=2.6cm, minimum height=1.3cm] at (11.05,-1.05) {};
  \node[cttl] at (12.35,-1.46) {Term Lost};
  \node[key] at (12.35,-1.96) {Wrong Answer};
