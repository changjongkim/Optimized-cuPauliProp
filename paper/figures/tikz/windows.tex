% Design, Fig. sectors (b): the sectors of two consecutive windows. Each window has its own valid rows (conserved by
% all its gates); two of them give four sectors (one color each). Inside a window, every sector runs alone. At the
% boundary, the output is split by the rows of the next window, so terms of different sectors (chip colors) meet in
% one sector of the next window.
\begin{tikzpicture}[x=1cm, y=1cm]
  \tikzset{chip/.style={draw=black!70, line width=0.3pt, minimum size=0.16cm, inner sep=0pt}}
  \newcommand{\gates}[3]{% x, prime, label
    \node[stxt, anchor=west] at (#1, -0.15) {#3};
    \foreach \i in {0,...,3}
      \node[cl, fill=gatec, minimum width=0.50cm, minimum height=0.30cm] at ({#1+0.52*\i}, -0.50) {};
    \node[stxt] at ({#1+0.25}, -0.50) {$Q#2_1$};
    \node[stxt] at ({#1+0.77}, -0.50) {$Q#2_2$};
    \node[stxt] at ({#1+1.29}, -0.50) {$Q#2_3$};
    \node[stxt] at ({#1+1.81}, -0.50) {$\cdots$};}
  \newcommand{\band}[4]{% x, row, color of the key, label
    \node[cl, fill=#3, minimum width=0.38cm, minimum height=0.30cm] at (#1, {-1.55-0.36*#2}) {#4};
    \draw[black!60, line width=0.4pt, fill=white] ({#1+0.42}, {-1.55-0.36*#2-0.15}) rectangle ++(1.70, 0.30);}
  % ---------------- window k
  \gates{0}{}{Gates of Window $k$}
  \node[dec, minimum width=2.10cm, font=\rmfamily\bfseries\scriptsize] at (1.03,-1.02) {Rows $r_1, r_2$ with\\$r\cdot Q_i = 0$ for All $i$};
  \foreach \c/\k/\n [count=\r from 0] in {sa/00/3, sb/01/2, sc/10/4, sd/11/3}
  {
    \band{0}{\r}{\c}{\k}
    \foreach \j in {1,...,\n}
      \node[chip, fill=\c] at ({0.25+0.30*\j}, {-1.55-0.36*\r}) {};
  }
  \node[sann] at (1.15,-3.08) {Merge and Truncate\\Inside Each Sector};
  % ---------------- the boundary
  \node[barrow, minimum height=0.55cm] at (2.65,-2.10) {};
  \node[stxt] at (2.65,-1.62) {Split};
  % ---------------- window k+1
  \gates{3.20}{'}{Gates of Window $k{+}1$}
  \node[dec, minimum width=2.10cm, font=\rmfamily\bfseries\scriptsize] at (4.23,-1.02) {Rows $r'_1, r'_2$ with\\$r'\cdot Q'_i = 0$ for All $i$};
  \foreach \c/\k [count=\r from 0] in {sa/00, sb/01, sc/10, sd/11}
    \band{3.20}{\r}{\c}{\k}
  % the terms of window k, regrouped: chip colors are their old sectors
  \foreach \r/\list in {0/{sa,sc,sd}, 1/{sb,sc,sa,sd}, 2/{sd,sb,sc}, 3/{sa,sc,sb,sd,sc}}
    \foreach \c [count=\j from 0] in \list
      \node[chip, fill=\c] at ({3.75+0.30*\j}, {-1.55-0.36*\r}) {};
  \node[sann] at (4.35,-3.08) {Terms of Different\\Sectors Meet};
\end{tikzpicture}
