% Design, Fig. store: host memory and one GPU during a window. Host memory holds the sectors of this window and of the
% next window as lists of fixed-size chunks taken from a pool. A GPU (1) loads one sector, (2) applies the gates of
% the window to it, (3) checks that the output fits its buffer, else splits the sector by one more valid row and
% continues with both halves, (4) partitions the output by the rows of the next window into chunks of the next
% window's sectors.
\begin{tikzpicture}[x=1cm, y=1cm]
  \tikzset{ch/.style={cl, minimum width=0.36cm, minimum height=0.22cm},
           vbar/.style={barrow, minimum height=0.36cm, rotate=-90},
           ubar/.style={barrow, minimum height=0.36cm, rotate=90}}
  % ---------------------------------------------------------------- host memory
  \draw[pan] (0,0) rectangle (13.90,-1.95);
  \node[ptitle, anchor=west] at (0.08,-0.25) {Host Memory of Node $d$};
  % this window
  \draw[comp] (0.15,-0.45) rectangle (4.95,-1.85);
  \node[ctitle, anchor=west] at (0.22,-0.63) {Sectors of This Window};
  \foreach \k/\c/\n/\x/\y in {00/sa/5/0.22/-0.90, 01/sb/3/0.22/-1.15, 10/sc/4/0.22/-1.40, 11/sd/2/0.22/-1.65}
  {
    \node[stxt, anchor=west] at (\x,\y) {\k};
    \foreach \j in {1,...,\n}
      \node[ch, fill=\c] at ({\x+0.18+0.37*\j},\y) {};
  }
  \node[sann, anchor=west, align=left] at (2.75,-1.28) {Each Sector:\\a List of\\Fixed-Size Chunks};
  % chunk pool
  \draw[comp] (5.20,-0.45) rectangle (8.70,-1.85);
  \node[ctitle, anchor=west] at (5.27,-0.63) {Chunk Pool};
  \foreach \j in {0,...,6}
    \node[free, minimum width=0.36cm, minimum height=0.22cm] at ({5.35+0.42*\j},-1.05) {};
  \node[sann] at (6.95,-1.55) {Pinned Once, Reused};
  % next window
  \draw[comp] (8.95,-0.45) rectangle (13.75,-1.85);
  \node[ctitle, anchor=west] at (9.02,-0.63) {Sectors of the Next Window};
  \foreach \k/\c/\n/\y in {00/sa/2/-0.90, 01/sb/3/-1.15, 10/sc/1/-1.40, 11/sd/2/-1.65}
  {
    \node[stxt, anchor=west] at (9.02,\y) {\k};
    \foreach \j in {1,...,\n}
      \node[ch, fill=\c] at ({9.20+0.37*\j},\y) {};
    \node[ch, draw=black!60, dashed, fill=white] at ({9.20+0.37*(\n+1)},\y) {};
  }
  \node[sann] at (12.45,-1.33) {Filled During\\the Window};
  % ---------------------------------------------------------------- GPU
  \draw[pan] (0,-2.30) rectangle (13.90,-4.75);
  \node[ptitle, anchor=west] at (0.08,-2.55) {GPU};
  \foreach \t/\c/\a/\b in {Input/sa/9.10/10.30, Output/white/10.30/11.50, Scratch/black!12/11.50/13.75}
  {
    \draw[draw=black!80, fill=\c, line width=0.4pt] (\a,-2.43) rectangle (\b,-2.73);
    \node[stxt] at ({(\a+\b)/2},-2.58) {\t};
  }
  \node[sann, anchor=east] at (9.05,-2.58) {GPU Memory};
  \node[act, minimum width=2.6cm] (apply) at (2.55,-3.40) {Apply the Window's Gates\\Branch, Merge, Truncate};
  \node[dec] (fits) at (6.45,-3.40) {Output Fits\\the Buffer?};
  \node[act, minimum width=2.6cm] (part) at (11.30,-3.40) {Split the Output by\\the Rows of the Next Window};
  \node[act] (half) at (6.45,-4.35) {Split the Sector by One More Valid Row};
  \draw[arr] (apply) -- (fits);
  \draw[arr] (fits) -- node[above, stxt] {Yes} (part);
  \draw[arr] (fits) -- node[right, stxt] {No} (half);
  \draw[dsh] (half.west) -| node[pos=0.75, left, sann] {Both Halves\\Continue} (apply.south);
  % between host memory and the GPU
  \node[vbar] at (2.55,-2.12) {};
  \node[ubar] at (11.30,-2.12) {};
  \node[num] at (2.15,-2.12) {1};
  \node[num] at ([xshift=-0.02cm, yshift=0.02cm]apply.north west) {2};
  \node[num] at ([xshift=-0.02cm, yshift=0.02cm]fits.north west) {3};
  \node[num] at (11.70,-2.12) {4};
\end{tikzpicture}
