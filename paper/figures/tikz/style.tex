% Style of the concept figures of the paper: the look of the design figures of the README (figures/src/tikz_style2.tex,
% after the ScaleQsim paper) at the printed size (full width: 13.9 cm), so the fonts are the printed fonts.
% Rules (2026-10-14):
%   bold        only the names of panels and components, and one key message per figure (red, italic)
%   regular     everything inside a component: table cells and headers, chips, action and decision boxes, labels
%   italic      short notes (no sentences; explanations belong in the caption and the text)
%   arrows      run through the gaps between components and end at their borders; text stays inside its own box
%   values      the running example only, no measured values
% colors: one palette file (palette_<name>.tex) defines every color name below; the default is set in standalone.tex
\input{\ifdefined\palette\palette\else palette_accent\fi.tex}
